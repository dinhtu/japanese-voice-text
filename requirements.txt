# Everything needed to run the pronunciation API + practice page and the tests.
#
#   python -m venv .venv
#   .venv\Scripts\activate          # macOS/Linux: source .venv/bin/activate
#   python -m pip install --upgrade pip
#   python -m pip install -r requirements.txt
#   uvicorn app.main:app --reload --host 0.0.0.0 --port 8000
#
# pyopenjtalk has no wheels on PyPI - it compiles from source on EVERY platform.
# Install a C/C++ toolchain + CMake first or pip dies on it:
#   Ubuntu/Debian  sudo apt update && sudo apt install -y python3.14-venv build-essential cmake python3-dev
#   Fedora/RHEL    sudo dnf install -y gcc gcc-c++ make cmake python3-devel
#   macOS          xcode-select --install  &&  brew install cmake
#   Windows        "Visual Studio Build Tools" + "Desktop development with C++"
#
# Linux x86_64: the PyPI torch wheel pulls ~16 nvidia-* CUDA packages (several GB)
# even with no GPU. On a CPU-only box, install torch first from the CPU index:
#   pip install torch torchaudio --index-url https://download.pytorch.org/whl/cpu
